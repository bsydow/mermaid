      domainstorytelling-beta
      group G_Service "Service Desk"
      group G_Engineering "Engineering"

      A_Customer "Customer" fa:fa-user in G_Service
      A_Agent "Support Agent" fa:fa-headset in G_Service
      A_Engineer "Engineer" fa:fa-laptop-code in G_Engineering
      A_Manager "Team Lead" fa:fa-user-tie in G_Engineering
      W_Ticket "Ticket" fa:fa-ticket-alt
      W_Backlog "Backlog" fa:fa-list
      W_Fix "Fix" fa:fa-wrench
      W_Release "Release" fa:fa-rocket

      A_Customer : 01 -- "reports" -> W_Ticket in G_Service -- "to" -> A_Agent id S_Report
      A_Agent : 02 -- "triages" -> W_Ticket in G_Service -- "into" -> W_Backlog in G_Service id S_Triage
      A_Agent : 03 -- "escalates" -> W_Ticket -- "to" -> A_Engineer id S_Escalate
      A_Engineer : 04 -- "implements" -> W_Fix in G_Engineering -- "for" -> W_Ticket in G_Engineering <- "reviews" -- A_Manager id S_Fix
      A_Engineer : 05 -- "ships" -> W_Fix in G_Engineering -- "in" -> W_Release in G_Engineering id S_Ship
      A_Agent : 05 -- "closes" -> W_Ticket in G_Service -- "and notifies" -> A_Customer id S_Close

      annotate actor A_Engineer "Owns root-cause analysis and the fix; loops in the team lead for review."

      annotate group G_Engineering "Second-level support. Only tickets L1 cannot resolve are escalated here."

      annotate sentence 01 "Entry point of the story — the seqNo badge 01 sits on the reporting arrow."

      annotate sentence S_Close "Referenced by id because seqNo 05 is reused (ship + close run in parallel)."

      annotate workobject W_Ticket@01 "The ticket instance as first created by the customer."

      annotate workobject W_Fix@S_Fix "The fix while under review, before it is bundled into a release."

Metropolis 1a: Going to the movies—coarse-grained—grouped by subdomain (Book Sample)

      ---
      title: Metropolis - 1a Going to the movies—coarse-grained—grouped by subdomain

      config:
        domainstorytelling:
          rankdir: TB
          nodeSpacing: 50
          rankSpacing: 100
          ranker: network-simplex
          acyclicer: greedy
      ---
      domainstorytelling-beta

      A_Cashier "Cashier" fa:fa-user
      A_Concession "Concession" fa:fa-user
      A_Moviegoer "Moviegoer" fa:fa-user
      A_Projectionist "Projectionist" fa:fa-user
      A_Usher "Usher" fa:fa-user
      W_Ticket "Ticket" fa:fa-ticket-alt
      W_SnackAndDrinks "Snack & Drinks" fa:fa-cocktail
      W_Movie "Movie" fa:fa-film

      group G_TicketSales "Ticket Sales" {
        A_Moviegoer : 01 -- "buys" -> W_Ticket -- "from" -> A_Cashier
      }

      group G_SnackSales "Snack Sales" {
        A_Moviegoer : 02 -- "buys" -> W_SnackAndDrinks -- "from" -> A_Concession
      }

      group G_Entrance "Entrance Control" {
        A_Moviegoer : 03 -- "shows" -> W_Ticket -- "to" -> A_Usher
        A_Usher : 04 -- "checks" -> W_Ticket -- "and grant entrance to" -> A_Moviegoer
      }

      group G_MovieProjection "Movie Projection" {
        A_Projectionist : 05 -- "starts" -> W_Movie -- "for" -> A_Moviegoer
        A_Moviegoer : 06 -- "watches" -> W_Movie
      }
